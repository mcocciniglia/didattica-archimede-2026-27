<?php
$attivita = [
    "robotica" => "Laboratorio di robotica",
    "teatro" => "Laboratorio di teatro"
];

if ($_SERVER["REQUEST_METHOD"] === "POST") {
    $datiValidi = true;
    $codice = "";
    $nome = "";
    $email = "";
    $etaTesto = "";
    $eta = false;

    // Il campo nascosto si controlla di nuovo sul server.
    if (!isset($_POST["attivita"])) {
        echo "<p>Parametro attivita non ricevuto. Riparti dalla scelta.</p>";
        $datiValidi = false;
    } elseif (!is_string($_POST["attivita"])) {
        echo "<p>Il codice dell’attività deve essere un testo.</p>";
        $datiValidi = false;
    } else {
        $codice = trim($_POST["attivita"]);
        if (!isset($attivita[$codice])) {
            echo "<p>Attività non disponibile. Riparti dalla scelta.</p>";
            $datiValidi = false;
        }
    }

    // Nome: presenza, tipo e testo non vuoto dopo trim.
    if (!isset($_POST["nome"])) {
        echo "<p>Parametro nome non ricevuto.</p>";
        $datiValidi = false;
    } elseif (!is_string($_POST["nome"])) {
        echo "<p>Il nome deve essere un testo.</p>";
        $datiValidi = false;
    } else {
        $nome = trim($_POST["nome"]);
        if ($nome === "") {
            echo "<p>Inserisci il nome: non bastano gli spazi.</p>";
            $datiValidi = false;
        }
    }

    // TODO 1: sostituisci le due righe provvisorie seguenti.
    // Email: isset, is_string, trim, vuoto e FILTER_VALIDATE_EMAIL.
    // Usa $email per il testo; ogni errore deve impostare false.
    echo "<p>Controllo email da completare.</p>";
    $datiValidi = false;

    // TODO 2: sostituisci le due righe provvisorie seguenti.
    // Età: isset, is_string, trim, vuoto e FILTER_VALIDATE_INT.
    // Usa $etaTesto e $eta; separa === false dai limiti 14–100.
    echo "<p>Controllo età da completare.</p>";
    $datiValidi = false;

    if ($datiValidi === true) {
        echo "<h2>Dati validi: riepilogo dell’iscrizione simulata</h2>";
        echo "<p>Attività: " . htmlspecialchars($attivita[$codice], ENT_QUOTES, 'UTF-8') . "</p>";
        echo "<p>Nome: " . htmlspecialchars($nome, ENT_QUOTES, 'UTF-8') . "</p>";
        // TODO 3: aggiungi email con htmlspecialchars e l’età intera validata.
        echo "<p>Simulazione conclusa: nessun dato è stato salvato.</p>";
    } else {
        echo "<p>Riepilogo non generato: correggi gli errori e riprova.</p>";
    }
} else {
    echo "<p>Apri la scelta dell’attività e invia il modulo con POST.</p>";
}
