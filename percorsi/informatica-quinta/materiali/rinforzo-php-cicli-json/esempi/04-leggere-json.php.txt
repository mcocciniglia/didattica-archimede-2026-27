<h1>Da JSON a PHP</h1>
<?php
$testo = '{"robotica":"Laboratorio di robotica","teatro":"Laboratorio di teatro"}';
$catalogo = json_decode($testo, true);

if (json_last_error() !== JSON_ERROR_NONE) {
    echo "<p>Il testo non è JSON valido.</p>";
} elseif (!is_array($catalogo)) {
    echo "<p>In questo esercizio serve un oggetto con codici e titoli.</p>";
} else {
    foreach ($catalogo as $codice => $titolo) {
        if (is_string($codice) && is_string($titolo)) {
            echo "<p>" . htmlspecialchars($codice, ENT_QUOTES, 'UTF-8');
            echo ": " . htmlspecialchars($titolo, ENT_QUOTES, 'UTF-8') . "</p>";
        } else {
            echo "<p>Ogni voce deve avere un codice e un titolo testuali.</p>";
        }
    }
}
?>
